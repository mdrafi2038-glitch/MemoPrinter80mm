    Bitmap renderMemo(Memo m){
        Bitmap src=BitmapFactory.decodeResource(getResources(), R.drawable.memo_template);
        if(src==null) throw new IllegalStateException("Memo template could not be loaded");
        final int w=576;
        final float sx=w/(float)src.getWidth();
        final int h=Math.round(src.getHeight()*sx);
        Bitmap b=Bitmap.createScaledBitmap(src,w,h,true).copy(Bitmap.Config.ARGB_8888,true);
        Canvas c=new Canvas(b);

        Paint p=new Paint(Paint.ANTI_ALIAS_FLAG | Paint.SUBPIXEL_TEXT_FLAG);
        p.setColor(Color.BLACK);
        p.setTypeface(memoRegular!=null?memoRegular:Typeface.create("sans-serif",Typeface.NORMAL));
        p.setStyle(Paint.Style.FILL);

        // 80mm print width: 576 dots keeps the full memo inside the common 80mm ESC/POS printable area and prevents side cropping.
        p.setTypeface(memoBold!=null?memoBold:Typeface.create("sans-serif",Typeface.BOLD));
        p.setTextSize(34*sx);
        draw(c,p,bn(String.valueOf(m.no)),270*sx,146*sx,Paint.Align.CENTER);
        draw(c,p,bn(m.date),520*sx,146*sx,Paint.Align.CENTER);

        // Customer information: customer name is bold and starts slightly right of the printed label.
        p.setTypeface(memoBold!=null?memoBold:Typeface.create("sans-serif",Typeface.BOLD));
        p.setTextSize(34*sx);
        draw(c,p,fit(m.name,360*sx,p),145*sx,216*sx,Paint.Align.LEFT);
        p.setTypeface(memoRegular!=null?memoRegular:Typeface.create("sans-serif",Typeface.NORMAL));
        draw(c,p,fit(m.address,360*sx,p),145*sx,279*sx,Paint.Align.LEFT);

        // Product rows: reserve the last TWO rows for discount and final total.
        float[] rows={418,479,540,600,660,720,781,841,902,962,1023};
        p.setTypeface(memoBold!=null?memoBold:Typeface.create("sans-serif",Typeface.BOLD));
        p.setTextSize(34*sx);
        int productCount=Math.min(m.items.size(), rows.length-2);
        for(int i=0;i<productCount;i++){
            Item x=m.items.get(i);
            float y=rows[i]*sx;
            draw(c,p,fit(x.name,275*sx,p),24*sx,y,Paint.Align.LEFT);
            draw(c,p,fit(x.qty,70*sx,p),356*sx,y,Paint.Align.CENTER);
            draw(c,p,fit(x.amount,145*sx,p),490*sx,y,Paint.Align.CENTER);
        }

        int discountRow=productCount;
        int totalRow=productCount+1;
        // Summary rows: labels stay in the middle/পরিমাণ column; values stay in the টাকা column.
        p.setTypeface(memoBold!=null?memoBold:Typeface.create("sans-serif",Typeface.BOLD));
        p.setTextSize(34*sx);
        draw(c,p,"ছাড়",356*sx,rows[discountRow]*sx,Paint.Align.CENTER);
        drawSummaryAmount(c,p,"-"+bn(formatNumber(m.discount)),rows[discountRow]*sx,sx);
        draw(c,p,"মোট",356*sx,rows[totalRow]*sx,Paint.Align.CENTER);
        drawSummaryAmount(c,p,bn(formatNumber(m.total)),rows[totalRow]*sx,sx);

        // Add a white safety tail below the signature boxes so the physical
        // printer has enough paper before the cutter is triggered.
        final int bottomPadding = 120;
        Bitmap padded=Bitmap.createBitmap(b.getWidth(), b.getHeight()+bottomPadding, Bitmap.Config.ARGB_8888);
        Canvas pc=new Canvas(padded);
        pc.drawColor(Color.WHITE);
        pc.drawBitmap(b,0,0,null);
        return padded;
    }
    void draw(Canvas c,Paint p,String s,float x,float y,Paint.Align a){p.setTextAlign(a);c.drawText(s==null?"":s,x,y,p);}
    void drawSummaryAmount(Canvas c,Paint p,String s,float y,float sx){
        // Keep discount/final amounts fully inside the 80mm memo's টাকা column.
        // Large totals are reduced only when necessary instead of being clipped.
        float original=p.getTextSize();
        float size=original;
        float maxWidth=154*sx;
        while(size>24*sx){
            p.setTextSize(size);
            if(p.measureText(s)<=maxWidth) break;
            size-=1*sx;
        }
        p.setTextAlign(Paint.Align.RIGHT);
        c.drawText(s==null?"":s,560*sx,y,p);
        p.setTextSize(original);
    }
    String fit(String s,float max,Paint p){if(s==null)return "";if(p.measureText(s)<=max)return s;String z=s;while(z.length()>1&&p.measureText(z+"…")>max)z=z.substring(0,z.length()-1);return z+"…";}

    void printMemos(ArrayList<Memo> memos){
        if(socket==null||!socket.isConnected()){
            toast("আগে Bluetooth printer connect করুন");
            printerScreen();
            return;
        }
        final ArrayList<Memo> batch=new ArrayList<>(memos);
        new Thread(()->{
            try{
                OutputStream out=socket.getOutputStream();
                for(int i=0;i<batch.size();i++){
                    Bitmap b=renderMemo(batch.get(i));
                    out.write(new byte[]{0x1B,0x40});
                    sendRaster(out,b);
                    out.write(new byte[]{0x1B,0x64,0x06});
                    out.write(new byte[]{0x1D,0x56,0x00});
                    out.flush();
                    if(i<batch.size()-1) Thread.sleep(500);
                }
                runOnUiThread(()->toast(bn(String.valueOf(batch.size()))+"টি marked memo একসাথে print হয়েছে ✓"));
            }catch(Exception e){
                runOnUiThread(()->toast("Bulk print failed — printer connection check করুন"));
            }
        }).start();
    }

    void sendRaster(OutputStream out,Bitmap b)throws IOException{int wb=(b.getWidth()+7)/8;int h=b.getHeight();out.write(new byte[]{0x1D,0x76,0x30,0x00,(byte)(wb&255),(byte)((wb>>8)&255),(byte)(h&255),(byte)((h>>8)&255)});byte[] row=new byte[wb];for(int y=0;y<h;y++){Arrays.fill(row,(byte)0);for(int x=0;x<b.getWidth();x++){int col=b.getPixel(x,y);int gray=(Color.red(col)*299+Color.green(col)*587+Color.blue(col)*114)/1000;if(gray<170)row[x>>3]|=(byte)(0x80>>(x&7));}out.write(row);if(y%64==0)out.flush();}}

        });

                toast("যে memoগুলো delete করতে চান সেগুলো আগে Mark করুন");
                return;
            }
            int count=marked.size();
            new AlertDialog.Builder(this)
                .setTitle("Marked Memo Delete")
                .setMessage(bn(String.valueOf(count))+"টি memo permanently delete করবেন?")
                .setNegativeButton("Cancel",null)
                .setPositiveButton("Delete",(d,w)->{
                    history.removeAll(marked);
                    marked.clear();
                    saveHistory();
                    toast("Marked memo delete হয়েছে ✓");

                }).show();
        });
