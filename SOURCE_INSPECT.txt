    @Override public void onBackPressed(){
        if(root==null){ super.onBackPressed(); return; }
        if(!backStack.isEmpty()){
            String previous=backStack.pop();
            restoringScreen=true;
            renderScreen(previous);
            restoringScreen=false;
        } else if(!"home".equals(currentScreen)){
            restoringScreen=true; home(); restoringScreen=false;
        } else {
            super.onBackPressed();
        }
    }

    void goBack(){ onBackPressed(); }

    void applyWindowInsets(View v){
        final int left=v.getPaddingLeft();
        final int top=v.getPaddingTop();
        final int right=v.getPaddingRight();
        final int bottom=v.getPaddingBottom();
        ViewCompat.setOnApplyWindowInsetsListener(v, (view, insets) -> {
            Insets bars=insets.getInsets(WindowInsetsCompat.Type.systemBars() | WindowInsetsCompat.Type.displayCutout());
            view.setPadding(left + bars.left, top + bars.top, right + bars.right, bottom + bars.bottom);
            return insets;
        });
        ViewCompat.requestApplyInsets(v);
    }

    ParsedInput parseAutomaticInput(String raw){
        ParsedInput out=new ParsedInput();
        String[] lines=raw==null?new String[0]:raw.split("\\r?\\n");
        ArrayList<String> clean=new ArrayList<>();
        for(String line:lines){String z=line.trim(); if(!z.isEmpty()) clean.add(z);}
        if(clean.size()<2){out.error="Customer এবং কমপক্ষে ১টি product দিন"; return out;}
        out.customer=phoneticBangla(clean.get(0));
        for(int i=1;i<clean.size();i++){
            String line=clean.get(i);
            if(line.startsWith("ঠিকানা:")){out.address=phoneticBangla(line.substring("ঠিকানা:".length()).trim()); continue;}
            if(line.toLowerCase(Locale.ROOT).startsWith("address:")){out.address=phoneticBangla(line.substring("address:".length()).trim()); continue;}
            int dash=line.lastIndexOf('-');
            if(dash<=0 || dash>=line.length()-1){out.error="এই format ভুল: "+line+" (Product-Quantity)"; return out;}
            String rawName=line.substring(0,dash).trim(); String qtyText=line.substring(dash+1).trim(); double qty=num(qtyText);
            if(rawName.isEmpty()||qty<=0){out.error="Product-Quantity ঠিকভাবে দিন: "+line; return out;}
            String bnName=phoneticBangla(rawName);
            Double price=findPriceAny(rawName,bnName);
            if(price==null){out.error="Default price নেই: "+rawName+"\nSettings-এ এই product-এর price save করুন"; return out;}
            out.items.add(new Item(bnName,bn(formatNumber(qty)),bn(formatNumber(price)),bn(formatNumber(qty*price))));
        }
        if(out.items.isEmpty()){out.error="কমপক্ষে ১টি product দিন"; return out;}
        if(out.items.size()>9){out.error="সর্বোচ্চ ৯টি product দিন"; return out;}
        return out;
    }

    Double findPriceAny(String raw,String converted){
        Double p=findPrice(raw); if(p!=null)return p;
        p=findPrice(converted); if(p!=null)return p;
        String nr=normalizeLatin(raw), nc=normalizeBangla(converted);
        for(String k:prices.keySet()){
            if(normalizeLatin(k).equals(nr) || normalizeBangla(k).equals(nc)) return prices.get(k);
        }
        return null;
    }

    String normalizeLatin(String s){return s==null?"":s.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]","");}
    String normalizeBangla(String s){return s==null?"":s.replaceAll("[ \u09cd\u09be-\u09cc\u09d7]","");}

    String phoneticBangla(String input){
        if(input==null||input.trim().isEmpty()) return "";
        String src=input.trim();
        boolean hasBangla=false;
        for(int i=0;i<src.length();i++){char ch=src.charAt(i); if(ch>='\u0980'&&ch<='\u09ff'){hasBangla=true;break;}}
        if(hasBangla) return src;
        StringBuilder result=new StringBuilder();
        String[] words=src.toLowerCase(Locale.ROOT).split("\\s+");
        for(int wi=0;wi<words.length;wi++){
            if(wi>0) result.append(' ');
            result.append(transliterateWord(words[wi]));
        }
        return result.toString();
    }

    String transliterateWord(String w){
        if(w==null||w.isEmpty()) return "";
        String special=SPECIAL_WORDS.get(w);
        if(special!=null) return special;
        // Common Bangla phonetic spellings used in order/memo names.
        String[][] whole={
            {"store","স্টোর"},{"shop","শপ"},{"traders","ট্রেডার্স"},{"trader","ট্রেডার"},
            {"pata","পাতা"},{"patak","পাতা কে"},{"masala","মসলা"},{"noodles","নুডলস"},
            {"pran","প্রাণ"},{"rafi","রাফি"},{"rahim","রহিম"},{"karim","করিম"},
            {"korian","করিয়ান"},{"korean","কোরিয়ান"},{"rice","রাইস"},{"oil","অয়েল"},
            {"salt","লবণ"},{"sugar","চিনি"},{"biscuit","বিস্কুট"},{"milk","মিল্ক"},{"k","কে"}
        };
        for(String[] x:whole) if(x[0].equals(w)) return x[1];
        return genericPhonetic(w);
    }

    String genericPhonetic(String s){
        // Lightweight offline phonetic transliteration. It preserves a normal
        // readable Bengali form and is intentionally editable after conversion.
        StringBuilder out=new StringBuilder(); int i=0; boolean afterConsonant=false;
        while(i<s.length()){
            if(Character.isDigit(s.charAt(i))){out.append(s.charAt(i));i++;continue;}
            String tri=i+3<=s.length()?s.substring(i,i+3):"";
            String bi=i+2<=s.length()?s.substring(i,i+2):"";
            String one=s.substring(i,i+1);
            String cluster=LATIN_CLUSTERS.get(tri);
            if(cluster!=null){out.append(cluster);i+=3;afterConsonant=true;continue;}
            cluster=LATIN_CLUSTERS.get(bi);
            if(cluster!=null){out.append(cluster);i+=2;afterConsonant=true;continue;}
            String vowel=VOWELS.get(one);
            if(vowel!=null){
                if(afterConsonant){out.append(vowel);}
                else out.append(vowel.length()==1&&vowel.equals("া")?"আ":vowel);
                i++; afterConsonant=false; continue;
            }
            String con=CONSONANTS.get(one);
            if(con!=null){out.append(con);i++;afterConsonant=true;continue;}
            out.append(s.charAt(i));i++;afterConsonant=false;
        }
        return out.toString();
    }

        if(src==null) throw new IllegalStateException("Memo template could not be loaded");

    void drawSummaryAmount(Canvas c,Paint p,String s,float y,float sx){
        // Keep discount/final amounts fully inside the 80mm memo's টাকা column.
        // Large totals are reduced only when necessary instead of being clipped.
        float original=p.getTextSize();
        float size=original;
        float maxWidth=154*sx;
        while(size>24*sx){
            p.setTextSize(size);
            if(p.measureText(s)<=maxWidth) break;
            size-=1*sx;
        }
        p.setTextAlign(Paint.Align.RIGHT);
        c.drawText(s==null?"":s,560*sx,y,p);
        p.setTextSize(original);
    }

    void printMemos(ArrayList<Memo> memos){
        if(socket==null||!socket.isConnected()){
            toast("আগে Bluetooth printer connect করুন");
            printerScreen();
            return;
        }
        final ArrayList<Memo> batch=new ArrayList<>(memos);
        new Thread(()->{
            try{
                OutputStream out=socket.getOutputStream();
                for(int i=0;i<batch.size();i++){
                    Bitmap b=renderMemo(batch.get(i));
                    out.write(new byte[]{0x1B,0x40});
                    sendRaster(out,b);
                    out.write(new byte[]{0x1B,0x64,0x06});
                    out.write(new byte[]{0x1D,0x56,0x00});
                    out.flush();
                    if(i<batch.size()-1) Thread.sleep(500);
                }
                runOnUiThread(()->toast(bn(String.valueOf(batch.size()))+"টি marked memo একসাথে print হয়েছে ✓"));
            }catch(Exception e){
                runOnUiThread(()->toast("Bulk print failed — printer connection check করুন"));
            }
        }).start();
    }

    void printMemo(Memo m){ if(socket==null||!socket.isConnected()){toast("আগে Bluetooth printer connect করুন");printerScreen();return;} final Memo toPrint=m;new Thread(()->{try{OutputStream out=socket.getOutputStream();out.write(new byte[]{0x1B,0x40});Bitmap b=renderMemo(toPrint);sendRaster(out,b);out.write(new byte[]{0x1B,0x64,0x06});out.write(new byte[]{0x1D,0x56,0x00});out.flush();runOnUiThread(()->toast("মেমো print হয়েছে ✓"));}catch(Exception e){runOnUiThread(()->toast("Print failed — printer connection check করুন"));}}).start(); }
    void sendRaster(OutputStream out,Bitmap b)throws IOException{int wb=(b.getWidth()+7)/8;int h=b.getHeight();out.write(new byte[]{0x1D,0x76,0x30,0x00,(byte)(wb&255),(byte)((wb>>8)&255),(byte)(h&255),(byte)((h>>8)&255)});byte[] row=new byte[wb];for(int y=0;y<h;y++){Arrays.fill(row,(byte)0);for(int x=0;x<b.getWidth();x++){int col=b.getPixel(x,y);int gray=(Color.red(col)*299+Color.green(col)*587+Color.blue(col)*114)/1000;if(gray<170)row[x>>3]|=(byte)(0x80>>(x&7));}out.write(row);if(y%64==0)out.flush();}}

    void savedMemos(){
        openScreen("saved");
        shell("Saved Memos","Local history • mark any memos and print together",true);
        ScrollView sv=scroll();
        LinearLayout c=content(sv);
        if(history.isEmpty()){
            c.addView(tv("কোনো saved memo নেই",15,MUTED));
            return;
        }

        // Bulk selection controls: the user can mark only the memos they want.
        LinearLayout tools=new LinearLayout(this);
        tools.setOrientation(LinearLayout.VERTICAL);
        LinearLayout selectRow=new LinearLayout(this);
        selectRow.setGravity(Gravity.CENTER_VERTICAL);
        TextView selected=tv("০টি selected",14,TEXT);
        selected.setTypeface(Typeface.DEFAULT,Typeface.BOLD);
        CheckBox all=new CheckBox(this);
        all.setText("সবগুলো Mark");
        all.setTextSize(13);
        all.setTextColor(TEXT);
        selectRow.addView(selected,new LinearLayout.LayoutParams(0,dp(52),1));
        selectRow.addView(all,new LinearLayout.LayoutParams(dp(125),dp(52)));
        tools.addView(selectRow);

        LinearLayout actionRow=new LinearLayout(this);
        Button bulkPrint=action("🖨  Marked Print");
        Button bulkDelete=lightAction("🗑  Marked Delete");
        actionRow.addView(bulkPrint,new LinearLayout.LayoutParams(0,dp(52),1));
        LinearLayout.LayoutParams deleteLp=new LinearLayout.LayoutParams(0,dp(52),1);
        deleteLp.setMargins(dp(8),0,0,0);
        actionRow.addView(bulkDelete,deleteLp);
        tools.addView(actionRow);
        c.addView(tools);
        c.addView(gap(10));

        final ArrayList<Memo> marked=new ArrayList<>();
        final ArrayList<CheckBox> boxes=new ArrayList<>();

        final Runnable[] updateSelection=new Runnable[1];
        updateSelection[0]=()->{
            selected.setText(bn(String.valueOf(marked.size()))+"টি selected");
            boolean checked=!history.isEmpty() && marked.size()==history.size();
            if(all.isChecked()!=checked){
                all.setOnCheckedChangeListener(null);
                all.setChecked(checked);
                all.setOnCheckedChangeListener((button, value)->{
                    marked.clear();
                    for(int i=0;i<history.size();i++){
                        CheckBox cb=boxes.get(i);
                        cb.setOnCheckedChangeListener(null);
                        cb.setChecked(value);
                        cb.setOnCheckedChangeListener((b,v)->{
                            Memo mm=(Memo)b.getTag();
                            if(v){ if(!marked.contains(mm)) marked.add(mm); }
                            else marked.remove(mm);
                            updateSelection[0].run();
                        });
                        if(value) marked.add(history.get(i));
                    }
                    updateSelection[0].run();
                });
            }
        };

        for(Memo m:history){
            LinearLayout r=card();
            LinearLayout top=new LinearLayout(this);
            top.setGravity(Gravity.CENTER_VERTICAL);
            CheckBox mark=new CheckBox(this);
            mark.setText("Mark");
            mark.setTextSize(13);
            mark.setTextColor(BLUE_DARK);
            mark.setTag(m);
            TextView h=tv("মেমো নং "+memoLabel(m.no),18,TEXT);
            h.setTypeface(Typeface.DEFAULT,Typeface.BOLD);
            top.addView(mark,new LinearLayout.LayoutParams(dp(75),dp(48)));
            top.addView(h,new LinearLayout.LayoutParams(0,dp(48),1));
            r.addView(top);
            r.addView(tv(m.date+"  •  "+(m.name.isEmpty()?"No customer":m.name),13,MUTED));
            r.addView(tv("পণ্য: "+bn(String.valueOf(m.items.size()))+"   •   মোট: ৳"+bn(formatNumber(m.total)),14,TEXT));

            ImageView thumb=new ImageView(this);
            thumb.setImageBitmap(renderMemo(m));
            thumb.setAdjustViewBounds(true);
            thumb.setScaleType(ImageView.ScaleType.FIT_CENTER);
            thumb.setBackgroundColor(WHITE);
            thumb.setContentDescription("Saved memo preview");
            LinearLayout.LayoutParams tp=new LinearLayout.LayoutParams(-1,dp(260));
            tp.setMargins(0,dp(8),0,dp(8));
            r.addView(thumb,tp);
            thumb.setOnClickListener(v->showSaved(m));

            mark.setOnCheckedChangeListener((button,checked)->{
                if(checked){ if(!marked.contains(m)) marked.add(m); }
                else marked.remove(m);
                updateSelection[0].run();
            });

            r.addView(gap(7));
            LinearLayout a=new LinearLayout(this);
            Button view=lightAction("View");
            view.setOnClickListener(v->showSaved(m));
            Button print=action("Print");
            print.setOnClickListener(v->printMemo(m));
            a.addView(view,new LinearLayout.LayoutParams(0,50,1));
            LinearLayout.LayoutParams pp=new LinearLayout.LayoutParams(0,50,1);
            pp.setMargins(8,0,0,0);
            a.addView(print,pp);
            r.addView(a);
            c.addView(r);
            c.addView(gap(10));
            boxes.add(mark);
        }

        all.setOnCheckedChangeListener((button,value)->{
            marked.clear();
            for(int i=0;i<history.size();i++){
                CheckBox cb=boxes.get(i);
                cb.setOnCheckedChangeListener(null);
                cb.setChecked(value);
                cb.setOnCheckedChangeListener((b,v)->{
                    Memo mm=(Memo)b.getTag();
                    if(v){ if(!marked.contains(mm)) marked.add(mm); }
                    else marked.remove(mm);
                    updateSelection[0].run();
                });
                if(value) marked.add(history.get(i));
            }
            updateSelection[0].run();
        });

        bulkPrint.setOnClickListener(v->{
            if(marked.isEmpty()){
                toast("যে memoগুলো print করতে চান সেগুলো আগে Mark করুন");
                return;
            }
            ArrayList<Memo> batch=new ArrayList<>(marked);
            printMemos(batch);
        });

        bulkDelete.setOnClickListener(v->{
            if(marked.isEmpty()){
                toast("যে memoগুলো delete করতে চান সেগুলো আগে Mark করুন");
                return;
            }
            int count=marked.size();
            new AlertDialog.Builder(this)
                .setTitle("Marked Memo Delete")
                .setMessage(bn(String.valueOf(count))+"টি memo permanently delete করবেন?")
                .setNegativeButton("Cancel",null)
                .setPositiveButton("Delete",(d,w)->{
                    history.removeAll(marked);
                    marked.clear();
                    saveHistory();
                    toast("Marked memo delete হয়েছে ✓");
                    refreshCurrentScreen();
                }).show();
        });
        updateSelection[0].run();
    }
