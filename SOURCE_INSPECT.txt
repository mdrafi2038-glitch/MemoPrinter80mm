    @Override public void onBackPressed(){
        if(root==null){ super.onBackPressed(); return; }
        if(!backStack.isEmpty()){
            String previous=backStack.pop();
            restoringScreen=true;
            renderScreen(previous);
            restoringScreen=false;
        } else if(!"home".equals(currentScreen)){
            restoringScreen=true; home(); restoringScreen=false;
        } else {
            super.onBackPressed();
        }
    }

    void goBack(){ onBackPressed(); }

    void applyWindowInsets(View v){
        final int left=v.getPaddingLeft();
        final int top=v.getPaddingTop();
        final int right=v.getPaddingRight();
        final int bottom=v.getPaddingBottom();
        ViewCompat.setOnApplyWindowInsetsListener(v, (view, insets) -> {
            Insets bars=insets.getInsets(WindowInsetsCompat.Type.systemBars() | WindowInsetsCompat.Type.displayCutout());
            view.setPadding(left + bars.left, top + bars.top, right + bars.right, bottom + bars.bottom);
            return insets;
        });
        ViewCompat.requestApplyInsets(v);
    }

    ParsedInput parseAutomaticInput(String raw){
        ParsedInput out=new ParsedInput();
        String[] lines=raw==null?new String[0]:raw.split("\\r?\\n");
        ArrayList<String> clean=new ArrayList<>();
        for(String line:lines){String z=line.trim(); if(!z.isEmpty()) clean.add(z);}
        if(clean.size()<2){out.error="Customer এবং কমপক্ষে ১টি product দিন"; return out;}
        out.customer=phoneticBangla(clean.get(0));
        for(int i=1;i<clean.size();i++){
            String line=clean.get(i);
            if(line.startsWith("ঠিকানা:")){out.address=phoneticBangla(line.substring("ঠিকানা:".length()).trim()); continue;}
            if(line.toLowerCase(Locale.ROOT).startsWith("address:")){out.address=phoneticBangla(line.substring("address:".length()).trim()); continue;}
            int dash=line.lastIndexOf('-');
            if(dash<=0 || dash>=line.length()-1){out.error="এই format ভুল: "+line+" (Product-Quantity)"; return out;}
            String rawName=line.substring(0,dash).trim(); String qtyText=line.substring(dash+1).trim(); double qty=num(qtyText);
            if(rawName.isEmpty()||qty<=0){out.error="Product-Quantity ঠিকভাবে দিন: "+line; return out;}
            String bnName=phoneticBangla(rawName);
            Double price=findPriceAny(rawName,bnName);
            if(price==null){out.error="Default price নেই: "+rawName+"\nSettings-এ এই product-এর price save করুন"; return out;}
            out.items.add(new Item(bnName,bn(formatNumber(qty)),bn(formatNumber(price)),bn(formatNumber(qty*price))));
        }
        if(out.items.isEmpty()){out.error="কমপক্ষে ১টি product দিন"; return out;}
        if(out.items.size()>9){out.error="সর্বোচ্চ ৯টি product দিন"; return out;}
        return out;
    }

    Double findPriceAny(String raw,String converted){
        Double p=findPrice(raw); if(p!=null)return p;
        p=findPrice(converted); if(p!=null)return p;
        String nr=normalizeLatin(raw), nc=normalizeBangla(converted);
        for(String k:prices.keySet()){
            if(normalizeLatin(k).equals(nr) || normalizeBangla(k).equals(nc)) return prices.get(k);
        }
        return null;
    }

    String normalizeLatin(String s){return s==null?"":s.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]","");}
    String normalizeBangla(String s){return s==null?"":s.replaceAll("[ \u09cd\u09be-\u09cc\u09d7]","");}

    String phoneticBangla(String input){
        if(input==null||input.trim().isEmpty()) return "";
        String src=input.trim();
        boolean hasBangla=false;
        for(int i=0;i<src.length();i++){char ch=src.charAt(i); if(ch>='\u0980'&&ch<='\u09ff'){hasBangla=true;break;}}
        if(hasBangla) return src;
        StringBuilder result=new StringBuilder();
        String[] words=src.toLowerCase(Locale.ROOT).split("\\s+");
        for(int wi=0;wi<words.length;wi++){
            if(wi>0) result.append(' ');
            result.append(transliterateWord(words[wi]));
        }
        return result.toString();
    }

    String transliterateWord(String w){
        if(w==null||w.isEmpty()) return "";
        String special=SPECIAL_WORDS.get(w);
        if(special!=null) return special;
        // Common Bangla phonetic spellings used in order/memo names.
        String[][] whole={
            {"store","স্টোর"},{"shop","শপ"},{"traders","ট্রেডার্স"},{"trader","ট্রেডার"},
            {"pata","পাতা"},{"patak","পাতা কে"},{"masala","মসলা"},{"noodles","নুডলস"},
            {"pran","প্রাণ"},{"rafi","রাফি"},{"rahim","রহিম"},{"karim","করিম"},
            {"korian","করিয়ান"},{"korean","কোরিয়ান"},{"rice","রাইস"},{"oil","অয়েল"},
            {"salt","লবণ"},{"sugar","চিনি"},{"biscuit","বিস্কুট"},{"milk","মিল্ক"},{"k","কে"}
        };
        for(String[] x:whole) if(x[0].equals(w)) return x[1];
        return genericPhonetic(w);
    }

    String genericPhonetic(String s){
        // Lightweight offline phonetic transliteration. It preserves a normal
        // readable Bengali form and is intentionally editable after conversion.
        StringBuilder out=new StringBuilder(); int i=0; boolean afterConsonant=false;
        while(i<s.length()){
            if(Character.isDigit(s.charAt(i))){out.append(s.charAt(i));i++;continue;}
            String tri=i+3<=s.length()?s.substring(i,i+3):"";
            String bi=i+2<=s.length()?s.substring(i,i+2):"";
            String one=s.substring(i,i+1);
            String cluster=LATIN_CLUSTERS.get(tri);
            if(cluster!=null){out.append(cluster);i+=3;afterConsonant=true;continue;}
            cluster=LATIN_CLUSTERS.get(bi);
            if(cluster!=null){out.append(cluster);i+=2;afterConsonant=true;continue;}
            String vowel=VOWELS.get(one);
            if(vowel!=null){
                if(afterConsonant){out.append(vowel);}
                else out.append(vowel.length()==1&&vowel.equals("া")?"আ":vowel);
                i++; afterConsonant=false; continue;
            }
            String con=CONSONANTS.get(one);
            if(con!=null){out.append(con);i++;afterConsonant=true;continue;}
            out.append(s.charAt(i));i++;afterConsonant=false;
        }
        return out.toString();
    }
